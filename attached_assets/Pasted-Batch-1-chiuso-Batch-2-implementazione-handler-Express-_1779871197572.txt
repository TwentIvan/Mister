Query param federationId opzionale (NULL/system default per default).
typescriptconst startTime = Date.now();

const config = await loadActiveConfig(db, req.query.federationId ?? null);

// Carica tutte le righe + stats (usa lo stesso JOIN che fa persist-voto-mister.ts)
const rows = await db.select({...}).from(playerGiornataStats)...;

// Calcola in memoria
const computed = rows.map(row => ({
  id: row.id,
  voto: computeVoti(row.statsJson, config).votoSynthesis,
}));

const toUpdate = computed.filter(c => c.voto !== null);
const skipped = computed.filter(c => c.voto === null).length;

// Bulk UPDATE con VALUES (stesso pattern già usato in persist-voto-mister.ts)
if (toUpdate.length > 0) {
  const values = toUpdate.map(c => `(${c.id}, ${c.voto})`).join(',');
  await db.execute(sql.raw(`
    UPDATE player_giornata_stats AS t
    SET voto_mister = v.voto::real
    FROM (VALUES ${values}) AS v(id, voto)
    WHERE t.id = v.id
  `));
}

res.json({
  updatedCount: toUpdate.length,
  skippedCount: skipped,
  durationMs: Date.now() - startTime,
});
