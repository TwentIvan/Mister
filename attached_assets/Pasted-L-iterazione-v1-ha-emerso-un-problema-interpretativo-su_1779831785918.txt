Esegui questa diagnostica:
sqlSELECT p.name, p.real_team, p.role_classic,
       (pgs.stats_json->'games'->>'minutes')::int AS min,
       (pgs.stats_json->'passes'->>'total')::int AS passes_total,
       (pgs.stats_json->'passes'->>'accuracy')::numeric AS passes_accuracy_raw,
       (pgs.stats_json->'passes'->>'key')::int AS passes_key,
       ROUND(
         (pgs.stats_json->'passes'->>'accuracy')::numeric * 100.0 / 
         NULLIF((pgs.stats_json->'passes'->>'total')::numeric, 0),
         1
       ) AS hypothesis_if_count
FROM player_giornata_stats pgs
JOIN players p ON p.id = pgs.player_id
WHERE (pgs.stats_json->'games'->>'minutes')::int = 90
  AND p.role_classic = 'MID'
ORDER BY (pgs.stats_json->'passes'->>'total')::int DESC
LIMIT 10;

passes_total (count totale tentati)
passes_accuracy_raw (valore grezzo come sta in JSON)
hypothesis_if_count (la percentuale che otterremmo se accuracy fosse un count)

Task 31 — Verifica anche sui 4 ruoli con statistica aggregata
sqlSELECT p.role_classic,
       COUNT(*) AS n,
       ROUND(AVG((stats_json->'passes'->>'total')::numeric), 1) AS avg_passes_total,
       ROUND(AVG((stats_json->'passes'->>'accuracy')::numeric), 1) AS avg_accuracy_raw,
       ROUND(AVG(
         (stats_json->'passes'->>'accuracy')::numeric * 100.0 /
         NULLIF((stats_json->'passes'->>'total')::numeric, 0)

FROM player_giornata_stats pgs
JOIN players p ON p.id = pgs.player_id

  AND (stats_json->'passes'->>'total')::int > 0
GROUP BY p.role_classic
ORDER BY p.role_classic;
