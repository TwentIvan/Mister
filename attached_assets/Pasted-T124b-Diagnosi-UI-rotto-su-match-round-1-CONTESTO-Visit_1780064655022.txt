2. ISPEZIONE LINEUP MARIO'S ROUND 1 SPECIFICO

  SELECT
    lp.player_id, p.full_name, p.role_classic,
    lp.is_starter, lp.bench_order, lp.is_captain,
    pgs.voto AS voto_round1
  FROM lineups l
  JOIN lineup_players lp ON lp.lineup_id = l.id
  JOIN players p ON p.id = lp.player_id
  LEFT JOIN player_giornata_stats pgs ON pgs.player_id = lp.player_id
    AND pgs.season = 2024 AND pgs.round = 1
  WHERE l.fanta_team_id = 'ft-mvp-1' AND l.season = 2024 AND l.round = 1
  ORDER BY lp.is_starter DESC, lp.bench_order ASC;
