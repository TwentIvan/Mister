Task 24 — Scrivere l'algoritmo
Crea scripts/voto-mister/compute.ts con questa logica per ogni riga player_giornata_stats:
typescripttype ComputedVoti = {
  votoStatsOnly: number | null;
  votoRatingStripped: number | null;
  votoBlend: number | null;
  votoSynthesis: number | null;  // canonico (= blend o fallback su stats)
};

function computeVoti(stats: StatsJson, config: VotoMisterConfig): ComputedVoti {
  const minutes = stats.games.minutes;
  
  // Soglia minuti
  if (minutes === null || minutes < config.minutes.threshold) {
    return { votoStatsOnly: null, votoRatingStripped: null, votoBlend: null, votoSynthesis: null };
  }
  
  // Fattore di estrapolazione (1.0 se min ≥ 60, capped at extrapolationCap se sotto)
  const extrap = minutes >= config.minutes.fullSample
    ? 1.0
    : Math.min(config.minutes.extrapolationCap, config.minutes.fullSample / minutes);
  
  // 1) voto_stats_only — somma contributi performance, con estrapolazione sui count
  const statsContrib = computeStatsContributions(stats, config.stats, extrap);
  const votoStatsOnly = config.anchor + statsContrib;
  
  // 2) voto_rating_stripped — api_rating con eventi rimossi (no estrapolazione)
  const rating = stats.games.rating !== null ? parseFloat(stats.games.rating) : null;
  const votoRatingStripped = rating === null ? null : rating - computeStripping(stats, config.stripping);
  
  // 3) voto_blend — α × stats + β × stripped (solo se entrambi disponibili)
  const votoBlend = votoRatingStripped === null
    ? null
    : config.blend.alphaStats * votoStatsOnly + config.blend.betaRating * votoRatingStripped;
  
  // 4) voto_synthesis — canonico: blend se disponibile, altrimenti fallback su stats_only
  const votoSynthesis = votoBlend !== null ? votoBlend : votoStatsOnly;
  
  return {
    votoStatsOnly: round2(votoStatsOnly),
    votoRatingStripped: round2(votoRatingStripped),
    votoBlend: round2(votoBlend),
    votoSynthesis: round2(votoSynthesis),
  };
}
Dove:

computeStatsContributions somma tutti i contributi di qualità e volume usando i pesi della config. L'estrapolazione si applica ai count (key_pass, tackle, ecc.) PRIMA del cap. Le percentuali non vengono moltiplicate (sono già rate).
computeStripping somma: 0.79 × goalFirst + 0.83 × max(0, gol-1) + 0.13 × penaltyScored + 0.60 × assist − 0.60 × penaltyMissed − 0.57 × penaltyCommitted − 1.0 × redCard + 1.0 × penaltySaved. Attenzione ai segni: lo stripping rimuove un premio (sottrae) o un malus (somma).
round2 arrotonda a 2 decimali.
